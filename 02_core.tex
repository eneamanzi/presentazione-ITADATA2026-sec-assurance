\begin{frame}
    \frametitle{Our Approach: Agnosticism and Contract-Driven}
    % Testo precedente:
    % - Application Agnosticism: No target-specific code. The target is described at runtime by its OpenAPI specification and configuration.
    % - Configuration as Source of Variation: Deployment-specific values and operational parameters are kept outside the assessment logic.
    % - Contract-Driven Probing: The specification defines the documented attack surface and guides the construction of valid requests.
    % Solo nello speech: esempi di valori di configurazione (path, credenziali per ruolo, soglie);
    % richieste valide = non scartate dalla validazione dell'input, arrivano alla logica applicativa.
    \begin{itemize}
        \setlength{\itemsep}{0.8em}
        \item \textbf{Application agnosticism:} no target-specific code; the target is described by its \alert{OpenAPI spec} and configuration

        \item \textbf{Configuration as source of variation:} all target-specific values live in \alert{one file}

        \item \textbf{Contract-driven testing:} the spec defines the \alert{attack surface} and guides the construction of \alert{valid requests}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Our Approach: Specified Oracles}

    % Testo precedente:
    % - Semantic Evaluation: Each test evaluates observed behavior against a specific security guarantee.
    % - 3-Step Oracle Construction: Identify the guarantee under verification. / Define what a violation looks like. / Encode the resulting evaluation rule.
    % - Epistemic Limit: Verdicts are only as sound as the contract they are checked against.
    % Solo nello speech: limite epistemico + la divergenza viene sempre segnalata, la gravità la decide un analista.
    % Esempi dalla Tabella 3 del paper, stessi domini della slide 6 (D1 black, D2 grey, D4 white).
    Every test has its own \alert{specified oracle}, built in three steps:

    \vspace{.6em}
    \begin{center}
        \textbf{1.} Identify the guarantee
        \quad$\rightarrow$\quad
        \textbf{2.} Define the \alert{violation}
        \quad$\rightarrow$\quad
        \textbf{3.} Encode the rule
    \end{center}

    \vspace{.8em}
    \uncover<2->{
            \begin{center}
                \renewcommand{\arraystretch}{1.3}
                \begin{tabular}{>{\addfontfeatures{Numbers=Lining}}l l >{\addfontfeatures{Numbers=Lining}}l}
                    \toprule
                    \textbf{Domain}   & \textbf{Criterion from} & \textbf{Violation}                             \\
                    \midrule
                    D1 Authentication & credential rules        & \alert{expired or revoked} credential accepted \\
                    D2 Authorization  & two roles compared      & user \alert{A reads user B}'s resource         \\
                    D4 Availability   & gateway configuration   & \alert{rate limit} not enforced                \\
                    \bottomrule
                \end{tabular}
            \end{center}
        }
\end{frame}

\begin{frame}
    \frametitle{Our Approach: Deterministic Assessment Engine}
    % Testo precedente:
    % - Dependency-Aware Scheduling: Resolves test dependencies into a deterministic execution order.
    % - Reproducible Assessment: Fixed configuration, ordering, and oracles make relevant results comparable across runs.
    % - Actionable Output: Structured results and semantic exit codes support automated CI/CD decisions.
    % Solo nello speech: esempio della revoca del token; 7 fasi (bloccanti / non bloccanti), cicli, tie-break lessicografico.
    \begin{itemize}
        \setlength{\itemsep}{0.8em}
        \item \textbf{Dependency-aware scheduling:} a test runs only after its \alert{prerequisites}

        \item \textbf{Reproducible assessment:} \alert{same configuration $\rightarrow$ same verdict}

        \item \textbf{Actionable output:} \alert{semantic exit codes} for CI/CD pipelines

    \end{itemize}
\end{frame}
