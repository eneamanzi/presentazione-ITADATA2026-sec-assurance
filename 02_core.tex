\begin{frame}
    \frametitle{Our Approach: Contract-Driven API Assurance}

    \textbf{Core Idea:} Derive valid probes from OpenAPI contracts and evaluate security behavior using specified, deterministic oracles.

    \begin{itemize}
        \item \textbf{Application Agnosticism \& Contract-Driven Probing} \\
              OpenAPI and configuration drive the assessment.

        \item \textbf{Specified Oracles for Security Guarantees} \\
              Explicit rules evaluate each security guarantee.

        \item \textbf{Eight-Domain Taxonomy \& Privilege Gradient} \\
              Security checks are organized by domain and access level.

        \item \textbf{Deterministic Assessment Engine} \\
              Test execution is coordinated for reproducible results.
    \end{itemize}

\end{frame}

\begin{frame}
    \frametitle{Approach Details: Application Agnosticism and Contract-Driven Probing}
    \begin{itemize}
        \item \textbf{Application Agnosticism:} No target-specific code. The target is described at runtime by its OpenAPI specification and configuration.

        \item \textbf{Configuration as Source of Variation:} Deployment-specific values and operational parameters are kept outside the assessment logic.

        \item \textbf{Contract-Driven Probing:} The specification defines the documented attack surface and guides the construction of valid requests.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Approach Details: Taxonomy and Privilege-Aware Coverage}
    \begin{itemize}
        \item \textbf{Eight-Domain Taxonomy (D0--D7):} Security guarantees are organized into semantically distinct domains.

        \item \textbf{Privilege-Aware Coverage:} Each test declares the access level it requires:
              \begin{itemize}
                  \item \textit{BLACK\_BOX:} Network access only.
                  \item \textit{GREY\_BOX:} Credentials for distinct roles.
                  \item \textit{WHITE\_BOX:} Access to the gateway configuration plane.
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Approach Details: Specified Oracles}
    \begin{itemize}
        \item \textbf{Semantic Evaluation:} Each test evaluates observed behavior against a specific security guarantee.

        \item \textbf{3-Step Oracle Construction:}
              \begin{enumerate}
                  \item Identify the guarantee under verification.
                  \item Define what a violation looks like.
                  \item Encode the resulting evaluation rule.
              \end{enumerate}

        \item \textbf{Epistemic Limit:} Verdicts are only as sound as the contract they are checked against.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Approach Details: Deterministic Assessment Engine}
    \begin{itemize}
        \item \textbf{Dependency-Aware Scheduling:} Resolves test dependencies into a deterministic execution order.

        \item \textbf{Reproducible Assessment:} Fixed configuration, ordering, and oracles make relevant results comparable across runs.

        \item \textbf{Actionable Output:} Structured results and semantic exit codes support automated CI/CD decisions.
    \end{itemize}
\end{frame}
