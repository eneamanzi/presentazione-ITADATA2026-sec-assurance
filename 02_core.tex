    \frametitle{Our Approach: Contract-Driven API Assurance}

    % High-level summary sentence (The Core Thesis)
    \textbf{Core Idea:} Derive syntactically valid probes from OpenAPI contracts and evaluate security semantics using deterministic, specified oracles.

    \begin{itemize}
        \item \textbf{Application Agnosticism \& Contract-Driven Probing} \\
              Zero target-specific code; input generation driven strictly by OpenAPI specs and configuration.

        \item \textbf{Specified Oracles for Security Guarantees} \\
              Domain-derived evaluation rules to detect semantic violations beyond HTTP status codes.

        \item \textbf{Eight-Domain Taxonomy \& Privilege Gradient} \\
              Structured test coverage scaling with available vantage points (Black, Grey, White Box).

        \item \textbf{Deterministic Assessment Engine} \\
              Topological scheduling and LIFO teardown ensuring 100\% reproducible verdicts for CI/CD.
    \end{itemize}

\end{frame}

\begin{frame}
    \frametitle{Approach Details: Agnosticism \& Contract-Driven}
    \begin{itemize}
        \item \textbf{Application Agnosticism:} Zero target-specific code. Logic is driven entirely by \texttt{openapi.yaml} and \texttt{config.yaml}.
        \item
        \item \textbf{Configuration as Source of Truth:} All deployment variables (URLs, credentials) are isolated from the engine.
        \item
        \item \textbf{Contract-Driven Probing:} The specification generates syntactically valid requests and defines the exact attack surface.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Approach Details: Taxonomy \& Privilege Gradient}
    \begin{itemize}
        \item \textbf{8-Domain Taxonomy (D0--D7):} Structured security guarantees (e.g., Auth, RBAC, Integrity) aligned with NIST standards.
        \item
        \item \textbf{Privilege-Aware Coverage (Box Gradient):} Coverage scales dynamically based on available access:
        \item
              \begin{itemize}
                  \item \textit{BLACK\_BOX:} Network-level perimeter controls.
                  \item \textit{GREY\_BOX:} Multi-role credentials for segregation (RBAC/BOLA).
                  \item \textit{WHITE\_BOX:} Gateway administrative configuration plane.
              \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Approach Details: Specified Oracles}
    \begin{itemize}
        \item \textbf{Semantic Evaluation:} Moves beyond basic syntax (e.g., HTTP 500 errors) to judge actual security behavior.
        \item
        \item \textbf{3-Step Oracle Construction:}
              \begin{enumerate}
                  \item Identify the specific security guarantee.
                  \item Define its behavioral violation.
                  \item Encode the evaluation rule.
              \end{enumerate}
        \item \textbf{Epistemic Limit:} Verdicts rely on contract compliance; an inaccurate specification yields flawed assumptions.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Approach Details: Deterministic Engine}
    \begin{itemize}
        \item \textbf{DAG-Based Scheduling:} Resolves test dependencies statically before execution (e.g., auth before access).
        \item \textbf{100\% Reproducible:} Fixed configuration and strict LIFO teardown guarantee identical outcomes across runs.
        \item \textbf{Actionable Output:} Distinct semantic exit codes route results directly to CI/CD pipelines.
    \end{itemize}
\end{frame}
