\begin{frame}
    \frametitle{Our Approach: Agnosticism and Contract-Driven}
    \begin{itemize}
        \setlength{\itemsep}{0.8em}
        \item \textbf{Application agnosticism:} no target-specific code; the target is described by its \alert{OpenAPI spec} and configuration

        \item \textbf{Configuration as source of variation:} all target-specific values live in \alert{one file}

        \item \textbf{Contract-driven testing:} the spec defines the \alert{attack surface} and guides the construction of \alert{valid requests}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Our Approach: Specified Oracles}

    Every test has its own \alert{specified oracle}, built in three steps:

    \vspace{.6em}
    \begin{center}
        \textbf{1.} Identify the guarantee
        \quad$\rightarrow$\quad
        \textbf{2.} Define the \alert{violation}
        \quad$\rightarrow$\quad
        \textbf{3.} Encode the rule
    \end{center}

    \vspace{.8em}
    \uncover<2->{
            \begin{center}
                \renewcommand{\arraystretch}{1.3}
                \begin{tabular}{>{\addfontfeatures{Numbers=Lining}}l l >{\addfontfeatures{Numbers=Lining}}l}
                    \toprule
                    \textbf{Domain}   & \textbf{Criterion from} & \textbf{Violation}                             \\
                    \midrule
                    D1 Authentication & credential rules        & \alert{expired or revoked} credential accepted \\
                    D2 Authorization  & two roles compared      & user \alert{A reads user B}'s resource         \\
                    D4 Availability   & gateway configuration   & \alert{rate limit} not enforced                \\
                    \bottomrule
                \end{tabular}
            \end{center}
        }
\end{frame}

\begin{frame}
    \frametitle{Our Approach: Deterministic Assessment Engine}

    \begin{itemize}
        \setlength{\itemsep}{0.8em}
        \item \textbf{Dependency-aware scheduling:} a test runs only after its \alert{prerequisites}

        \item \textbf{Reproducible assessment:} \alert{same configuration $\rightarrow$ same verdict}

        \item \textbf{Actionable output:} \alert{semantic exit codes} for CI/CD pipelines

    \end{itemize}
\end{frame}
