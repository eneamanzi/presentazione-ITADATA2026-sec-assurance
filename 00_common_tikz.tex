\tikzset{
	pics/hyperarc between fan in/.style n args={5}{
		code={
			\path let \p1 = ($(#1)!.5!(#2)$), \p2 = ($(#2)!.5!(#3)$) in
        		coordinate (#4) at (\x1, \y2);
   			\foreach \src in {#1, #2} {
        		\draw[-, hyperarc, #5] (\src) -- (#4);
    		}
    		\draw[->, hyperarc, #5] (#4) -- (#3);
		}
	},
		pics/hyperarc between fan out/.style n args={5}{
		code={
			\path let \p1 = ($(#1)!.5!(#2)$), \p2 = ($(#2)!.5!(#3)$) in
        		coordinate (#4) at (\x1, \y2);

   			\foreach \src in {#1, #2} {
        		\draw[<-, hyperarc, #5] (\src) -- (#4);
    		}
    		\draw[-, hyperarc, #5] (#4) -- (#3);
		}
	},
}
