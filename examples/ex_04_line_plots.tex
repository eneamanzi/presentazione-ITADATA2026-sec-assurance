% ESEMPIO 4: frase che si completa a passi, due grafici a linee affiancati, riga di sintesi.
%   - La frase "... varying the number of" è visibile ai passi 1 e 2 e viene completata al passo 2:
%     due \only<...> di seguito nello stesso paragrafo.
%   - Grafici a linee con segni (mark) e tratteggi diversi per ogni serie, così si distinguono anche
%     stampati in bianco e nero. I dati sono in file .dat letti con \pgfplotstableread (la prima riga
%     è l'intestazione delle colonne; x index e y index scelgono le colonne).
%   - I due grafici stanno in una tabella a due colonne, con la didascalia sotto ciascuno.
%   - $\Longrightarrow$ \alert{...}: la conclusione, in una riga, sotto i grafici.

\begin{frame}
    \frametitle{Example: completed sentence, line plots, summary line}

    \pgfmathsetmacro\numberPreStart{int(1)}
    \pgfmathsetmacro\numberPreEnd{int(\numberPreStart+0)}
    \pgfmathsetmacro\numberGraphStart{int(\numberPreEnd+1)}
    \pgfmathsetmacro\numberGraphEnd{int(\numberGraphStart+0)}

    \begin{overlayarea}{\textwidth}{.9\textheight}

        \only<\numberPreStart-\numberGraphEnd>{
            \alert{Measure} varying the number of }
        \only<\numberGraphStart-\numberGraphEnd>{%
            \alert{parameter A} and \alert{parameter B}
        }

        \only<\numberPreStart-\numberPreEnd>{

            \vspace{1em}

            \begin{itemize}\setlength{\itemsep}{1em}
                \item \alert{parameter A} in $\{$2, \ldots, 5$\}$
                \item \alert{parameter B} in $\{$1, \ldots, 5$\}$
            \end{itemize}

        }

        \only<\numberGraphStart-\numberGraphEnd>{

            \vspace{1em}

            \pgfplotstableread{examples/lines_01.dat}\dataTableVaryingA
            \pgfplotstableread{examples/lines_02.dat}\dataTableVaryingB

            \pgfplotsset{every tick label/.append style={font=\normalsize}}

            \begin{adjustbox}{max totalsize={.99\textwidth}{.6\textheight},center}
                \begin{tabular}{cc}
                    \begin{tikzpicture}[scale=.5]
                        \begin{axis}[
                                xlabel={\normalsize \textbf{Parameter A}},
                                ylabel={\normalsize \textbf{Measure}},
                                xmin=2, xmax=5,
                                xtick={2, ..., 5},
                                ymin=4, ymax=11,
                                legend pos=north west,
                                legend columns=2,
                                legend style={/tikz/every even column/.append style={column sep=0.5cm}},
                                scale only axis
                            ]
                            \addplot[color=black, mark options={solid}]
                            table [x index=0, y index=1] {\dataTableVaryingA};
                            \addlegendentry{1 unit}

                            \addplot[color=black, mark options={solid}, dashed]
                            table [x index=0, y index=2] {\dataTableVaryingA};
                            \addlegendentry{2 units}

                            \addplot[color=black, mark options={solid}, dotted]
                            table [x index=0, y index=3] {\dataTableVaryingA};
                            \addlegendentry{3 units}

                            \addplot[color=black, mark options={solid}, dashdotted]
                            table [x index=0, y index=4] {\dataTableVaryingA};
                            \addlegendentry{4 units}

                            \addplot[color=black, mark options={solid},
                                dash pattern={on 1pt off 2pt on 1pt off 2pt on 3pt}]
                            table [x index=0, y index=5] {\dataTableVaryingA};
                            \addlegendentry{5 units}
                        \end{axis}
                    \end{tikzpicture} &

                    \begin{tikzpicture}[scale=.5]
                        \begin{axis}[
                                xlabel={\normalsize \textbf{Parameter B}},
                                xmin=1, xmax=5,
                                xtick={1,2,3,4,5},
                                ymin=4, ymax=11,
                                legend pos=north west,
                                scale only axis
                            ]
                            \addplot[color=black, mark options={solid}]
                            table [x index=0, y index=1] {\dataTableVaryingB};
                            \addlegendentry{2 items}

                            \addplot[color=black, dashed, mark options={solid}]
                            table [x index=0, y index=2] {\dataTableVaryingB};
                            \addlegendentry{3 items}

                            \addplot[color=black, dashdotted, mark options={solid}]
                            table [x index=0, y index=3] {\dataTableVaryingB};
                            \addlegendentry{4 items}

                            \addplot[color=black, dotted, mark options={solid}]
                            table [x index=0, y index=4] {\dataTableVaryingB};
                            \addlegendentry{5 items}
                        \end{axis}
                    \end{tikzpicture} \\
                    {\footnotesize Varying parameter A} & {\footnotesize Varying parameter B}
                \end{tabular}
            \end{adjustbox}

            \vspace{1em}

            $\Longrightarrow$ \alert{linear} increase of the measure

        }

    \end{overlayarea}

\end{frame}
