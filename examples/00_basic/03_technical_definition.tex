\begin{frame}
    \frametitle{03 Basic: defining a component}

    \begin{block}{Component definition}
        A \alert{component} is represented by
        \alert{$\angled{n, I, O}$}, where

        \vspace{.75em}

        \begin{itemize}\setlength{\itemsep}{.5em}
            \item \alert{$n$} is the component name;
            \item \alert{$I$} is the set of input elements;
            \item \alert{$O$} is the set of output elements.
        \end{itemize}
    \end{block}
\end{frame}
