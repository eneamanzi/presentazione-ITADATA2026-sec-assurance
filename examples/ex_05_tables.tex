%     Le colonne di testo lungo hanno la larghezza di un testo di riferimento: p{\widthof{testo}}.
%     L'intestazione è in colore evidenziato: \textbf{\alert{...}}.
%   Seconda slide: tabella con intestazione su due righe (\multirow e \multicolumn) accanto a un elenco con
%     formule, in due colonne.
%   In entrambe: booktabs (\toprule, \midrule, \bottomrule) e adjustbox per ridurre la tabella alla slide.

\begin{frame}
    \frametitle{Example: stacked tables with captions}

    \begin{adjustbox}{max totalsize={\textwidth}{\textheight},center}
        \begin{tabular}{c}
            \begin{tabular}{l c l l}
                \toprule
                \textbf{\alert{Name}} & \textbf{\alert{Column 2}} & \textbf{\alert{Column 3}} & \textbf{\alert{Description}}\\
                \midrule
                \texttt{Item1} & A & B--C & First item\\
                \texttt{Item2} & A & B & Second item\\
                \texttt{Item3} & C & C & Third item\\
                \bottomrule
            \end{tabular}\\
            \textbf{First table}\\
            ~\\[1em]
            \begin{tabular}{p{\widthof{\texttt{Item4}, \texttt{Item5}\ }} c l p{\widthof{A longer description}}}
                \toprule
                \textbf{\alert{Name}} & \textbf{\alert{Column 2}} & \textbf{\alert{Column 3}} & \textbf{\alert{Description}}\\
                \midrule
                \texttt{Item4}, \texttt{Item5} & A & B & A longer description of both items\\
                \texttt{Item6} & C & B & Another long description that wraps on two lines\\
                \bottomrule
            \end{tabular}\\
            \textbf{Second table}\\
        \end{tabular}
    \end{adjustbox}

\end{frame}
