\begin{frame}
    \frametitle{Context: Big Data APIs \& Security Assurance}

    % Goal della slide: Stabilire il contesto (Big Data API) e far capire subito dove si colloca il problema.
    % Testo precedente (spostato nello speech):
    % - Big Data Ecosystem: distributed services organized into pipelines and exposed through REST APIs
    % - Assurance Gap: infrastructure and functional testing leave API security semantics largely unchecked
    % Esempi solo nello speech: tenant che legge la pipeline di un altro tenant (BOLA);
    % endpoint dichiarato protetto che risponde senza credenziali (anticipa il risultato chiave).
    \begin{itemize}
        \item \textbf{Big Data Ecosystem:} distributed services exposed through \alert{APIs} and protected by API \alert{gateways}
              \pause

        \item \textbf{Trust Boundary:} APIs and gateways control access to pipelines, data, and services
              \pause

        \item \textbf{Assurance Gap:} syntactic and functional checks do not prove the expected \alert{security behavior}
    \end{itemize}

    \vspace{0.4em}
    \begin{center}
        \textbf{Examples of \alert{behavior} violations}

        \vspace{0.4em}
        \begin{tabular}{rcl}
            User A requests User B's resource
             & $\rightarrow$ & unauthorized access \\[0.4em]
            Expired or revoked credential
             & $\rightarrow$ & request accepted
        \end{tabular}
    \end{center}

\end{frame}

\begin{frame}
    \frametitle{State of the Art Limitations and Objectives}

    % Gap del paper: G1 = riga 1, G2 = riga 2, G3 (portabilità vs profondità) + G4 (flakiness) = riga 3.
    % Testo precedente:
    % - Syntactic Blindness: Checks request/response syntax, but not security semantics
    % - The Oracle Problem: No automated criterion to evaluate security guarantees
    % - Unreliable Execution: Flaky results make continuous CI/CD integration unreliable
    % - Semantic Verification: Verify security behavior beyond request/response syntax
    % - Specified Oracles: Define explicit rules for evaluating security guarantees
    % - Agnostic & Deterministic: Build a reproducible engine, portable across any documented target
    \begin{columns}[T]
        \begin{column}{.5\textwidth}
            \begin{block}{State of the Art Limitations}
                \begin{itemize}
                    \item<1-> \textbf{Syntactic Blindness:} \\
                          valid requests can still violate security
                    \item<2-> \textbf{The Oracle Problem:} \\
                          no criterion to judge a violation
                    \item<3-> \textbf{Shallow \& Flaky Tools:} \\
                          tools trade depth for reach, results vary across runs
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{.5\textwidth}
            % Objectives
            \begin{block}{Our Objectives}
                \begin{itemize}
                    \item<1-> \textbf{Semantic Verification:} \\
                          check security \alert{behavior}, not only syntax
                    \item<2-> \textbf{Specified Oracles:} \\
                          an \alert{explicit rule} for each guarantee
                    \item<3-> \textbf{Agnostic \& Deterministic:} \\
                          \alert{portable} across targets, \alert{reproducible} across runs
                \end{itemize}
            \end{block}
        \end{column}

    \end{columns}

\end{frame}
