\begin{frame}
    \frametitle{Context: Big Data APIs \& Security Assurance}

    % Goal della slide: Stabilire il contesto (Big Data API) e far capire subito dove si colloca il problema.
    \begin{itemize}
        \item \textbf{Big Data Ecosystem:} distributed services organized into pipelines and exposed through REST APIs

        \item \textbf{Trust Boundary:} API Gateways centralize main security controls.

        \item \textbf{Assurance Gap:} infrastructure and functional testing leave API security semantics largely unchecked

        \item \textbf{Consequence:} An API-level weakness can expose the pipelines and data behind the interface.
    \end{itemize}

\end{frame}

\begin{frame}
    \frametitle{State of the Art Limitations and Objectives}

    \begin{columns}[T]
        \begin{column}{.5\textwidth}
            \begin{block}{State of the Art Limitations}
                \begin{itemize}
                    \item \textbf{Syntactic Blindness:} \\
                          Checks request/response syntax, but not security semantics
                    \item \textbf{The Oracle Problem:} \\
                          No automated criterion to evaluate security guarantees
                    \item \textbf{Unreliable Execution:} \\
                          Flaky results make continuous CI/CD integration unreliable
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{.5\textwidth}
            % Objectives
            \begin{block}{Our Objectives}
                \begin{itemize}
                    \item \textbf{Semantic Verification:} \\
                          Verify security behavior beyond request/response syntax
                    \item \textbf{Specified Oracles:} \\
                          Define explicit rules for evaluating security guarantees
                    \item \textbf{Agnostic \& Deterministic:} \\
                          Build a reproducible engine, portable across any documented target.
                \end{itemize}
            \end{block}
        \end{column}

    \end{columns}

\end{frame}
